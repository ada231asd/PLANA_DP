import re

from pathlib import Path
from urllib.parse import quote, urljoin, urlparse

from bs4 import BeautifulSoup
from playwright.sync_api import sync_playwright

from config import (
    MGEKO_BASE,
    SEARCH_URL,
    HEADLESS,
    BROWSER,
    NAVIGATION_TIMEOUT_MS,
    DOWNLOAD_TIMEOUT_MS,
    IMAGE_RETRIES,
)

from models import Manga, Chapter


class MgekoError(Exception):
    pass


class MgekoClient:
    def __init__(self, log=None):
        self.log = log or (lambda _: None)

        self.pw = None
        self.browser = None
        self.context = None
        self.page = None

    # ==========================================================
    # BROWSER
    # ==========================================================

    def start(self):
        self.log("Запускаю браузер...")

        self.pw = sync_playwright().start()

        browser_name = BROWSER.lower().strip()

        if browser_name == "firefox":
            self.log("Браузер: Firefox")

            self.browser = self.pw.firefox.launch(
                headless=HEADLESS,
            )

        else:
            self.log("Браузер: Chromium")

            self.browser = self.pw.chromium.launch(
                headless=HEADLESS,
                args=[
                    "--disable-blink-features=AutomationControlled",
                ],
            )

        self.context = self.browser.new_context(
            viewport={
                "width": 1440,
                "height": 1000,
            },
            user_agent=(
                "Mozilla/5.0 (Windows NT 10.0; Win64; x64; rv:131.0) "
                "Gecko/20100101 Firefox/131.0"
            ),
            locale="en-US",
            timezone_id="America/Toronto",
        )

        self.context.set_default_timeout(NAVIGATION_TIMEOUT_MS)

        self.page = self.context.new_page()

        # ------------------------------------------------------
        # Логирование ошибок страницы
        # ------------------------------------------------------

        self.page.on(
            "console",
            lambda message: self.log(f"[JS] {message.type}: {message.text}"),
        )

        self.page.on(
            "pageerror",
            lambda error: self.log(f"[PAGE ERROR] {error}"),
        )

        self.page.on(
            "requestfailed",
            lambda request: self.log(
                f"[REQUEST FAILED] {request.method} {request.url} -> {request.failure}"
            ),
        )

        self.page.on(
            "response",
            self._on_response,
        )

        self.log("Браузер запущен.")

    def _on_response(self, response):
        """
        Показываем в лог только потенциально интересные
        ответы, чтобы не засорять консоль сотнями запросов.
        """

        url = response.url

        if "mgeko.cc" in url and response.status >= 400:
            self.log(f"[HTTP {response.status}] {url}")

    # ==========================================================
    # CLOSE
    # ==========================================================

    def close(self):
        debug_keep_browser = False

        if debug_keep_browser:
            self.log("DEBUG: браузер оставлен открытым.")

            try:
                while self.page and not self.page.is_closed():
                    time.sleep(1)
            except KeyboardInterrupt:
                pass

            return

        try:
            if self.context:
                self.context.close()
        except Exception as exc:
            self.log(f"Ошибка закрытия context: {exc}")

        try:
            if self.browser:
                self.browser.close()
        except Exception as exc:
            self.log(f"Ошибка закрытия browser: {exc}")

        try:
            if self.pw:
                self.pw.stop()
        except Exception as exc:
            self.log(f"Ошибка остановки Playwright: {exc}")

        self.page = None
        self.context = None
        self.browser = None
        self.pw = None

    # ==========================================================
    # NAVIGATION
    # ==========================================================

    def _goto(self, url):
        if not self.page:
            raise MgekoError("Браузер не запущен.")

        self.log("")
        self.log("=" * 70)
        self.log(f"Открываю страницу:")
        self.log(url)
        self.log("=" * 70)

        try:
            response = self.page.goto(
                url,
                wait_until="domcontentloaded",
                timeout=NAVIGATION_TIMEOUT_MS,
            )

            if response:
                self.log(f"HTTP статус: {response.status}")
                self.log(f"Финальный URL: {self.page.url}")

        except Exception as exc:
            self.log(f"Ошибка перехода на страницу: {exc}")
            self._save_debug_page("navigation_error")
            raise

        try:
            self.page.wait_for_load_state(
                "networkidle",
                timeout=8_000,
            )
            self.log("Состояние networkidle достигнуто.")

        except Exception:
            self.log("networkidle не достигнут (это не обязательно ошибка).")

        try:
            self.page.wait_for_timeout(1500)
        except Exception:
            pass

        self.log(f"Текущий URL: {self.page.url}")
        self.log(f"Заголовок страницы: {self.page.title()}")

    # ==========================================================
    # HELPERS
    # ==========================================================

    @staticmethod
    def _clean(text):
        return re.sub(r"\s+", " ", text or "").strip()

    @staticmethod
    def _title_from_url(url):
        slug = url.rstrip("/").split("/")[-1]
        return re.sub(r"[-_]+", " ", slug).title()

    @staticmethod
    def _normalize_manga_url(href):
        """
        Приводит любой URL к каноническому виду страницы тайтла:

            /manga/handsomely-monstrous            -> .../manga/handsomely-monstrous/
            /manga/handsomely-monstrous/           -> .../manga/handsomely-monstrous/
            /manga/handsomely-monstrous/?a=1       -> .../manga/handsomely-monstrous/
            https://www.mgeko.cc/manga/foo/bar/    -> None (слишком глубоко)

        Возвращает строку или None, если это не страница тайтла.
        """
        if not href:
            return None

        absolute = urljoin(MGEKO_BASE, href.strip())
        parsed = urlparse(absolute)

        # Хост должен быть mgeko.cc (или поддомен)
        host = parsed.netloc.lower()
        if "mgeko.cc" not in host:
            return None

        parts = [p for p in parsed.path.split("/") if p]

        # Нас интересует ровно /manga/<slug> — 2 сегмента
        if len(parts) != 2 or parts[0] != "manga":
            return None

        slug = parts[1].strip()

        # Отсекаем служебные разделы
        if not slug or slug.lower() in {
            "all-chapters",
            "page",
            "list",
            "latest",
            "hot",
        }:
            return None

        return f"{MGEKO_BASE}/manga/{slug}/"

    # ==========================================================
    # SEARCH
    # ==========================================================

    def search(self, query):
        query = self._clean(query)

        if not query:
            return []

        url = f"{SEARCH_URL}?inputContent={quote(query)}"

        self.log("")
        self.log(f"Поиск: {query}")
        self.log(f"URL поиска: {url}")

        self._goto(url)

        # Сохраняем страницу для диагностики.
        self._save_debug_page("search_page")
        self.save_debug_screenshot("search_page")

        html = self.page.content()

        self.log(f"Получено HTML: {len(html):,} символов")

        soup = BeautifulSoup(html, "html.parser")

        # ------------------------------------------------------
        # Информация о странице
        # ------------------------------------------------------

        html_title = self._clean(soup.title.get_text() if soup.title else "")
        self.log(f"HTML title: {html_title}")

        # ------------------------------------------------------
        # Собираем все /manga/<slug>/ и нормализуем
        # ------------------------------------------------------

        candidates = {}  # normalized_url -> {"title": str, "url": str}

        for a in soup.select("a[href]"):
            href = a.get("href", "")
            if not href:
                continue

            normalized = self._normalize_manga_url(href)
            if not normalized:
                continue

            # --- Заголовок: ссылка → alt картинки → title → slug ---
            text = self._clean(a.get_text(" ", strip=True))

            if not text:
                img = a.select_one("img[alt]")
                if img:
                    text = self._clean(img.get("alt", ""))

            if not text:
                text = self._clean(a.get("title", ""))

            if not text:
                text = self._title_from_url(normalized)

            existing = candidates.get(normalized)
            if existing is None:
                candidates[normalized] = {
                    "title": text,
                    "url": normalized,
                }
            elif len(text) > len(existing["title"] or ""):
                # Нашли более информативный текст — обновляем
                existing["title"] = text

        candidates_list = list(candidates.values())

        self.log(f"Уникальных страниц /manga/: {len(candidates_list)}")

        for i, item in enumerate(candidates_list[:50], start=1):
            self.log(f"  {i:02d}. {item['title']} -> {item['url']}")

        # ------------------------------------------------------
        # Скоринг релевантности
        # ------------------------------------------------------

        query_lower = query.lower()
        words = [w for w in re.split(r"\s+", query_lower) if w]
        slug_query = query_lower.replace(" ", "-")

        scored = []

        for item in candidates_list:
            title_lower = (item["title"] or "").lower()
            url_lower = item["url"].lower()

            score = 0

            # Точное совпадение названия
            if title_lower == query_lower:
                score += 1000
            elif query_lower in title_lower:
                score += 500
            elif title_lower and title_lower in query_lower:
                score += 200

            # Совпадение со слагом в URL
            if slug_query in url_lower:
                score += 300

            # Пословные совпадения
            for word in words:
                if word in title_lower:
                    score += 50
                if word in url_lower:
                    score += 20

            if score > 0:
                scored.append((score, item))

        scored.sort(key=lambda v: (-v[0], v[1]["title"].lower()))

        results = [item for _, item in scored]

        # Если фильтрация ничего не дала — отдаём всё, что нашли
        if not results:
            self.log(
                "По запросу нет точных совпадений. Возвращаю все найденные тайтлы."
            )
            results = candidates_list

        results = results[:30]

        self.log("")
        self.log(f"ИТОГОВЫХ РЕЗУЛЬТАТОВ: {len(results)}")

        for i, item in enumerate(results, start=1):
            self.log(f"[RESULT {i}] {item['title']} -> {item['url']}")

        return results

    # ==========================================================
    # MANGA PREVIEW (быстро, без рендера JS)
    # ==========================================================

    def get_manga_preview(self, url):
        """
        Лёгкий разбор страницы тайтла: обложка, статус, жанры, число глав.
        Работает через HTTP-контекст Playwright (без рендера JS) — быстро.
        Возвращает dict или None.
        """
        try:
            response = self.context.request.get(
                url,
                headers={"Referer": MGEKO_BASE},
                timeout=NAVIGATION_TIMEOUT_MS,
            )

            if not response.ok:
                self.log(f"Превью {url}: HTTP {response.status}")
                return None

            html = response.text()

        except Exception as exc:
            self.log(f"Ошибка превью {url}: {exc}")
            return None

        soup = BeautifulSoup(html, "html.parser")

        title = self._text_first(
            soup,
            [
                ".novel-title",
                "h1.novel-title",
                "h1",
                "meta[property='og:title']",
            ],
        ) or self._title_from_url(url)

        cover = ""

        img = soup.select_one(".novel-info img[src], .novel-cover img[src], img[src]")

        if img:
            cover = urljoin(MGEKO_BASE, img.get("src", ""))

        status = self._extract_status(soup)

        genres = []

        for a in soup.select("a[href]"):
            href = a.get("href", "").lower()
            if "genre" in href:
                text = self._clean(a.get_text(" ", strip=True))
                if text and text not in genres:
                    genres.append(text)

        chapters_count = len(soup.select(".chapter-list-item"))

        return {
            "title": title,
            "url": url,
            "cover_url": cover,
            "status": status,
            "genres": genres,
            "chapters_count": chapters_count,
        }

    # ==========================================================
    # SEARCH + PREVIEW
    # ==========================================================

    def search_with_previews(self, query, limit=12):
        """
        Возвращает результаты поиска + превью для первых `limit` штук.
        Использовать, когда UI хочет карточки, а не просто список строк.
        """
        results = self.search(query)

        enriched = []

        for i, item in enumerate(results):
            if i >= limit:
                enriched.append(item)
                continue

            preview = self.get_manga_preview(item["url"])

            if preview:
                # сохраняем "title" из поиска, если превью вернуло пустое
                if not preview.get("title"):
                    preview["title"] = item["title"]

                enriched.append(preview)
            else:
                enriched.append(item)

        return enriched

    def get_manga(self, url):
        self._goto(url)

        self._save_debug_page("manga_page")
        self.save_debug_screenshot("manga_page")

        soup = BeautifulSoup(self.page.content(), "html.parser")

        title = self._text_first(

        if not title:
            title = self._title_from_url(url)

        alternative = self._text_first(soup, [".alternative-title"])
        author = self._extract_labeled(soup, "Author")
        status = self._extract_status(soup)

        description = self._text_first(
            soup,
            [
                ".description",
                ".summary",
                ".novel-info .intro",
                ".intro",
            ],
        )

        # ------------------------------------------------------
        # Cover
        # ------------------------------------------------------

        cover = ""

        img = soup.select_one(".novel-info img[src], .novel-cover img[src], img[src]")

        if img:
            cover = urljoin(MGEKO_BASE, img.get("src", ""))

        # ------------------------------------------------------
        # Stats
        # ------------------------------------------------------

        views = self._extract_stat(soup, ["views", "view"])
        bookmarks = self._extract_stat(soup, ["bookmarked", "bookmark"])

        # ------------------------------------------------------
        # Genres
        # ------------------------------------------------------

        genres = []

        for a in soup.select("a[href]"):
            text = self._clean(a.get_text(" ", strip=True))
            href = a.get("href", "")

            if not text:
                continue

            if "genre" in href.lower() or "genres" in href.lower():
                if text not in genres:
                    genres.append(text)

        # ------------------------------------------------------
        # Chapters
        # ------------------------------------------------------

        chapters = self._parse_chapters(soup)

        self.log(f"Название: {title}")
        self.log(f"Автор: {author}")
        self.log(f"Статус: {status}")
        self.log(f"Глав найдено: {len(chapters)}")

        # ------------------------------------------------------
        # Full chapter list
        # ------------------------------------------------------

        all_link = soup.select_one("#library-push[href]")

        if all_link:
            all_url = urljoin(MGEKO_BASE, all_link.get("href", ""))

            if "/all-chapters/" in all_url:
                try:
                    self.log("Открываю полный список глав:")
                    self.log(all_url)

                    self._goto(all_url)

                    all_soup = BeautifulSoup(
                        self.page.content(),
                        "html.parser",
                    )

                    all_chapters = self._parse_chapters(all_soup)

                    self.log(f"В полном списке найдено: {len(all_chapters)}")

                    chapters = self._merge_chapters(chapters, all_chapters)

                except Exception as exc:
                    self.log("Не удалось загрузить полный список глав: " + str(exc))

        # Возвращаемся на страницу manga.
        try:
            self._goto(url)
        except Exception:
            pass

        return Manga(
            title=title,
            url=url,
            alternative_title=alternative,
            author=author,
            status=status,
            description=description,
            cover_url=cover,
            views=views,
            bookmarks=bookmarks,
            genres=genres,
            chapters=chapters,
        )

    # ==========================================================
    # CHAPTER PARSER
    # ==========================================================

    def _parse_chapters(self, soup):
        chapters = []
        seen = set()

        nodes = soup.select(".chapter-list-item")

        self.log(f"Элементов .chapter-list-item: {len(nodes)}")

        for node in nodes:
            number = node.get("data-chapterno", "").strip()

            a = node.select_one("a[href]")
            if not a:
                continue

            href = a.get("href", "")
            if not href:
                continue

            if "/reader/" not in href:
                continue

            chapter_url = urljoin(MGEKO_BASE, href)
            key = chapter_url.rstrip("/")

            if key in seen:
                continue

            chapter_number_node = node.select_one(".chapter-number")

            if chapter_number_node:
                label = self._clean(chapter_number_node.get_text(" ", strip=True))
            else:
                label = self._clean(node.get_text(" ", strip=True))

            stats_node = node.select_one(".chapter-stats")

            if stats_node:
                updated = self._clean(stats_node.get_text(" ", strip=True))
            else:
                updated = ""

            # Если data-chapterno отсутствует — тянем из URL.
            if not number:
                match = re.search(
                    r"chapter-"
                    r"([^/]+?)"
                    r"(?:-eng)?"
                    r"(?:-[^/]*)?/?$",
                    href,
                    re.I,
                )
                if match:
                    number = match.group(1)

            if not number:
                number = "?"

            try:
                order = float(number.replace(",", "."))
            except ValueError:
                order = 0

            seen.add(key)

            chapter = Chapter(
                number=number,
                url=chapter_url,
                label=label,
                updated=updated,
                order=order,
            )

            chapters.append(chapter)

            self.log(f"Глава {number}: {chapter_url}")

        chapters.sort(
            key=lambda chapter: (chapter.order, chapter.number),
            reverse=True,
        )

        return chapters

    # ==========================================================
    # MERGE CHAPTERS
    # ==========================================================

    @staticmethod
    def _merge_chapters(first, second):
        merged = {}

        for chapter in first + second:
            key = chapter.url.rstrip("/")
            merged[key] = chapter

        result = list(merged.values())

        result.sort(
            key=lambda chapter: (chapter.order, chapter.number),
            reverse=True,
        )

        return result

    def get_chapter_images(self, chapter_url):
        self._goto(chapter_url)

        self._save_debug_page("chapter_page")
        self.save_debug_screenshot("chapter_page")

        soup = BeautifulSoup(self.page.content(), "html.parser")

        images = []
        seen = set()

        # ------------------------------------------------------
        # Основной вариант
        # ------------------------------------------------------

        for img in soup.select("img[src]"):
            src = img.get("src", "").strip()
            if not src:
                continue

            absolute = urljoin(chapter_url, src)
            low = absolute.lower()

            if "credits-mgeko.png" in low:
                continue

            if "/sv2/comic/" not in low:
                continue

            image_id = img.get("id", "")
            match = re.search(r"image-(\d+)$", image_id)

            if match:
                numeric = int(match.group(1))
            else:
                numeric = 999999

            if absolute in seen:
                continue

            seen.add(absolute)
            images.append((numeric, absolute))

        # ------------------------------------------------------
        # Сортировка
        # ------------------------------------------------------

        images.sort(key=lambda item: item[0])

        # ------------------------------------------------------
        # Fallback
        # ------------------------------------------------------

        if not images:
            self.log("Основной поиск изображений ничего не нашёл.")

            for img in soup.select("img[src*='/sv2/comic/']"):
                src = img.get("src", "").strip()
                if not src:
                    continue

                if "credits-mgeko.png" in src.lower():
                    continue

                absolute = urljoin(chapter_url, src)

                if absolute in seen:
                    continue

                seen.add(absolute)
                images.append((len(images) + 1, absolute))

        result = [url for _, url in images]

        self.log(f"Найдено изображений: {len(result)}")

        for index, image_url in enumerate(result, start=1):
            self.log(f"{index:03d}: {image_url}")

        return result

    # ==========================================================

    # ==========================================================

    def download_image(self, url, destination, referer):
        last_error = None

        for attempt in range(1, IMAGE_RETRIES + 1):
            try:
                self.log(f"Скачивание {attempt}/{IMAGE_RETRIES}: {url}")

                response = self.context.request.get(
                    url,
                    headers={
                        "Referer": referer,
                        "Accept": (
                            "image/avif,"
                            "image/webp,"
                            "image/apng,"
                            "image/svg+xml,"
                            "image/*,"

    @staticmethod
    def _text_first(soup, selectors):
        for selector in selectors:
            node = soup.select_one(selector)

            if not node:
                continue

            if node.name == "meta":
                value = node.get("content", "")
            else:
                value = node.get_text(" ", strip=True)

            value = re.sub(r"\s+", " ", value or "").strip()

            if value:
                return value

        return ""

    def _extract_labeled(self, soup, label):
        text = soup.get_text(" ", strip=True)

        pattern = (
            rf"{re.escape(label)}"
            rf"\s*:\s*"
            rf"([^|]+?)"
            rf"(?=\s+"
            rf"(?:Status|Views|"
            rf"Bookmark|Genres|Author)"
            rf"\b|$)"
        )

        match = re.search(pattern, text, re.I)

        if match:
            return self._clean(match.group(1))

        return ""

        value = self._extract_labeled(soup, "Status")

            r"Complete|"
            r"On Going"
            r")\b",
            text,

        )

        if match:

                rf"\s*{word}\b",

import re

from PySide6.QtCore import QObject, Signal, Slot

def safe_name(value):

    value = re.sub(
        r'[<>:"/\\|?*]',
        "_",
        value,
    )

    value = re.sub(
        r"\s+",
        " ",
        value,
    ).strip()

    return value[:150] or "Untitled"


class DownloadWorker(QObject):
    progress = Signal(
        int,
        int,
        str,
    )

    chapter_started = Signal(str)

    log = Signal(str)

    error = Signal(str)

    finished = Signal()

    def __init__(
        self,
        manga_title,
        chapters,
    ):
        super().__init__()

        self.manga_title = manga_title
        self.chapters = chapters

        self._stop = False

    def stop(self):
        self._stop = True

    @Slot()
    def run(self):

        client = MgekoClient(self.log.emit)

        try:
            client.start()

            total_chapters = len(self.chapters)

            # -------------------------------------------------
            # CHAPTERS
            # -------------------------------------------------

            for (
                chapter_index,
                chapter,
            ) in enumerate(
                self.chapters,
                start=1,
            ):
                if self._stop:
                    break

                self.chapter_started.emit(f"Глава {chapter.number}")

                self.log.emit(f"Получаю страницы главы {chapter.number}...")

                image_urls = client.get_chapter_images(chapter.url)

                if not image_urls:
                    self.error.emit(f"В главе {chapter.number} не найдены изображения.")

                    continue

                # -------------------------------------------------
                # FOLDER
                # -------------------------------------------------

                chapter_dir = (
                    Path(DOWNLOAD_DIR)
                    / safe_name(self.manga_title)
                    / ("Chapter " + safe_name(chapter.number))
                )

                chapter_dir.mkdir(
                    parents=True,
                    exist_ok=True,
                )

                count = len(image_urls)

                # -------------------------------------------------
                # IMAGES
                # -------------------------------------------------

                for (
                    image_index,
                    image_url,
                ) in enumerate(
                    image_urls,
                    start=1,
                ):
                    if self._stop:
                        break

                    suffix = (
                        Path(
                            image_url.split(
                                "?",
                                1,
                            )[0]
                        ).suffix.lower()
                        or ".jpg"
                    )

                    destination = chapter_dir / f"{image_index:03d}{suffix}"

                    self.progress.emit(
                        chapter_index - 1,
                        total_chapters,
                        (f"Глава {chapter.number}: {image_index}/{count}"),
                    )

                    client.download_image(
                        image_url,
                        destination,
                        chapter.url,
                    )

                self.progress.emit(
                    chapter_index,
                    total_chapters,
                    (f"Глава {chapter.number}: готово"),
                )

        except Exception as exc:
            self.error.emit(str(exc))

        finally:
            client.close()

            self.finished.emit()

    result = Signal(object)
    cover = Signal(bytes)

    error = Signal(str)
    log = Signal(str)
    finished = Signal()

    def __init__(
        self,
        url,
    ):
        super().__init__()

        self.url = url

    @Slot()
    def run(self):

        client = MgekoClient(self.log.emit)

        try:
            client.start()

            manga = client.get_manga(self.url)

            self.result.emit(manga)

            # ---------------------------------------------
            # COVER
            # ---------------------------------------------

            if manga.cover_url:
                try:
                    response = client.context.request.get(
                        manga.cover_url,
                        headers={"Referer": manga.url},
                        timeout=30_000,
                    )

                    if response.ok:
                        self.cover.emit(response.body())

                except Exception as exc:
                    self.log.emit(f"Не удалось загрузить обложку: {exc}")

        except Exception as exc:
            self.error.emit(str(exc))

        finally:
            client.close()

            self.finished.emit()

from PySide6.QtCore import QObject, Signal, Slot

from core.mgeko import MgekoClient


class SearchWorker(QObject):
    results = Signal(object)
    error = Signal(str)
    log = Signal(str)
    finished = Signal()

    def __init__(self, query, enrich=True, preview_limit=12):
        super().__init__()

        self.query = query
        self.enrich = enrich
        self.preview_limit = preview_limit

    @Slot()
    def run(self):
        client = MgekoClient(self.log.emit)

        try:
            client.start()

            if self.enrich:
                results = client.search_with_previews(
                    self.query,
                    limit=self.preview_limit,
                )
            else:
                results = client.search(self.query)

            self.results.emit(results)

        except Exception as exc:
            self.error.emit(str(exc))

        finally:
            client.close()
            self.finished.emit()
